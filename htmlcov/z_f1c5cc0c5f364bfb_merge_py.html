126def _stop_merge_map(stops: pd.DataFrame, tolerance: float, crs: Any | None) -> dict[str, str]: 

127 """Map every fused stop_id onto the representative that survives. 

128 

129 The intent is the one the OSM public-transport builder applies to platforms 

130 -- points close enough together are one place -- but not its mechanism. That 

131 builder snaps to a one-metre grid, which is adequate when the coordinates 

132 come from a single survey and platforms all but coincide. Across agencies the 

133 same kerb is published tens of metres apart, and at that tolerance a grid 

134 stops describing distance: whether two stops fuse would depend on where the 

135 cell boundary happens to fall rather than on how far apart they are. 

136 

137 So the rule here is an explicit radius, assigned greedily in ``stop_id`` 

138 order: each stop either joins the first representative within ``tolerance`` 

139 or becomes one itself. Two properties follow, and both matter. Every fused 

140 stop really is within the tolerance of the stop it was fused into. And the 

141 result cannot chain -- under a transitive rule a line of stops each just 

142 inside the tolerance of its neighbour would collapse end to end, however long 

143 the line. 

144 """ 

145 points = _stop_points(stops) 

146 if points.empty: 

147 return {} 

148 

149 target = ( 

150 CRS.from_user_input(crs) 

151 if crs is not None 

152 else estimate_crs_for_bounds( 

153 float(points["_lon"].min()), 

154 float(points["_lat"].min()), 

155 float(points["_lon"].max()), 

156 float(points["_lat"].max()), 

157 ) 

158 ) 

159 if not target.is_projected: 159 ↛ 160line 159 didn't jump to line 160 because the condition on line 159 was never true

160 raise ValueError(f"merge_stops_within needs a projected CRS in metres, got {target.to_string()}") 

161 

162 projected = gpd.GeoSeries(gpd.points_from_xy(points["_lon"], points["_lat"]), crs="EPSG:4326").to_crs(target) 

163 

164 # Sorting makes the surviving identifier a property of the data rather than 

165 # of the order the feeds happened to be passed in. 

166 order = np.argsort(points["stop_id"].astype(str).to_numpy(), kind="stable") 

167 stop_ids = points["stop_id"].astype(str).to_numpy()[order] 

168 coords = np.column_stack([projected.x.to_numpy()[order], projected.y.to_numpy()[order]]) 

169 

170 tree = cKDTree(coords) 

171 assigned = np.full(len(stop_ids), -1, dtype="int64") 

172 mapping: dict[str, str] = {} 

173 for index in range(len(stop_ids)): 

174 if assigned[index] != -1: 

175 continue 

176 assigned[index] = index 

177 for neighbour in tree.query_ball_point(coords[index], tolerance): 

178 if assigned[neighbour] == -1: 

179 assigned[neighbour] = index 

180 mapping[stop_ids[neighbour]] = stop_ids[index] 

181 return mapping

224 """Combine several GTFS feeds into one that the graph builder can consume. 

225 

226 Identifiers are qualified by their source, so a ``route_id`` of ``1`` in two

230 Args: 

231 sources: Paths to GTFS directories or ZIP archives, or already-read 

232 feeds. A single source is returned unchanged unless stop merging is 

233 requested, so this function is a no-op where merging is not needed. 

234 prefixes: One prefix per source. Defaults to each source's file name, 

235 with a numeric suffix where two sources share one. 

236 merge_stops_within: Distance in metres below which stops published by 

237 different feeds are treated as the same place and fused into one

239 changes the topology of the network, and doing it silently would make 

240 an inferred transfer indistinguishable from a published one. Values 

241 around 15--25 m suit feeds from different agencies; the 1 m used for 

242 OSM platforms is far too tight here, because two agencies survey the

251
