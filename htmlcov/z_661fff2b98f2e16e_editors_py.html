132# Packed coordinate ids: latitude and longitude in units of 1e-7 degree (the precision of OSM), shifted to be 

133# non-negative and combined as lat * LON_SPAN + lon, which stays below 2**63. 

134_COORD_SCALE = 10_000_000 

135_LAT_SHIFT = 90 * _COORD_SCALE 

136_LON_SHIFT = 180 * _COORD_SCALE 

137_LON_SPAN = 2 * _LON_SHIFT + 1

140def coordinate_node_ids(points: gpd.GeoSeries | gpd.GeoDataFrame) -> np.ndarray: 

141 """Stable int64 node ids packed from the coordinates of points. 

142 

143 The longitude and latitude in EPSG:4326 are rounded to 1e-7 degree (the precision of OpenStreetMap), shifted 

144 to be non-negative and combined as ``(lat + 90°) * 3_600_000_001 + (lon + 180°)`` in units of 1e-7 degree, so 

145 ``divmod(node_id, 3_600_000_001)`` gives them back. The ids are below ``2**63`` and never negative. 

146 

147 Args: 

148 points: Points with a CRS, for example ``graph.nodes_gdf``. 

149 

150 Returns: 

151 Array of int64 ids in the order of ``points``. 

152 

153 Raises: 

154 ValueError: If the points have no CRS. 

155 """ 

156 if isinstance(points, gpd.GeoDataFrame): 156 ↛ 157line 156 didn't jump to line 157 because the condition on line 156 was never true

157 points = points.geometry 

158 if points.crs is None: 158 ↛ 159line 158 didn't jump to line 159 because the condition on line 158 was never true

159 raise ValueError("coordinate node ids need points with a CRS") 

160 lonlat = points.to_crs(4326) if not points.crs.equals("EPSG:4326") else points 

161 lon = np.rint(lonlat.x.to_numpy() * _COORD_SCALE).astype(np.int64) + _LON_SHIFT 

162 lat = np.rint(lonlat.y.to_numpy() * _COORD_SCALE).astype(np.int64) + _LAT_SHIFT 

163 return lat * _LON_SPAN + lon

166def _hashed_node_ids(nodes: gpd.GeoDataFrame, coordinate_ids: np.ndarray, shared: np.ndarray) -> np.ndarray: 

167 """Negative ids of nodes that share a coordinate id, from that id, their attributes and their rank among equals.""" 

168 columns = sorted((c for c in nodes.columns if c != nodes.geometry.name), key=str) 

169 rows = nodes.loc[shared, columns].itertuples(index=False, name=None) if columns else iter(()) 

170 keys = [ 

171 f"{cid}|{json.dumps(list(row), default=str, ensure_ascii=False)}" 

172 for cid, row in zip(coordinate_ids[shared], rows if columns else ([] for _ in range(int(shared.sum())))) 

173 ] 

174 rank = pd.Series(keys).groupby(keys, sort=False).cumcount().to_numpy() 

175 ids = np.empty(len(keys), dtype=np.int64) 

176 for i, (key, r) in enumerate(zip(keys, rank)): 

177 digest = hashlib.blake2b(f"{key}|{r}".encode(), digest_size=8).digest() 

178 ids[i] = -(int.from_bytes(digest, "big") & (2**63 - 1)) - 1 

179 return ids

188 """ 

189 Relabel node ids and update edge endpoints. 

190 

191 The function does not mutate the source graph. For multigraphs, ``k`` is 

192 recalculated after relabeling to keep ``['u', 'v', 'k']`` unique. 

193 

194 Args: 

195 graph_gdf: Source graph.

199 graph is built again from newer data, and graphs of neighbouring territories get 

200 equal ids for their shared nodes (see :func:`coordinate_node_ids`). 

201 on_duplicate: What ``by="coordinates"`` does with nodes closer than 1e-7 degree (about

206 is built again in the same way; the other nodes keep their coordinate ids. 

207 

208 Raises: 

209 ValueError: If ``by="coordinates"`` and the graph has no CRS, or nodes share a

436def project_objects2urban_graph(

443) -> tuple[UrbanGraphChanges, pd.Series]: 

444 """Prepare graph changes that connect objects to their nearest edges. 

445 

446 The function does not mutate the source graph. It computes nodes and edges to 

447 add, source edges to remove, and an ``object2node`` mapping. In backend services 

448 this is the preferred low-level workflow: ``changes`` can be persisted to graph 

449 tables and ``object2node_map`` can be saved on the object table as 

450 ``graph_node_id``. For in-memory workflows, apply the result with 

451 :func:`apply_urban_graph_changes`. 

452 

453 Objects are projected by their ``representative_point()``, so the function works 

454 for buildings, services and other polygon or point layers with a stable unique 

455 index. 

456 

457 Args: 

458 graph_gdf: Source urban graph. 

459 objects_gdf: Objects to connect to the graph. The index should be a stable 

460 object identifier, such as ``object_id`` or ``building_id``. 

461 speed_m_per_min: Movement speed on created connector edges, in meters per 

462 minute. For 5 km/h use ``5 * 1000 / 60``. 

463 max_dist: Optional maximum distance from an object to the nearest edge. If 

464 ``None``, nearest edges are searched without a distance limit. 

465 add_link_edge: If ``True``, create a dedicated object node and connector 

466 edge. If ``False``, map the object to the projection node on the graph.
