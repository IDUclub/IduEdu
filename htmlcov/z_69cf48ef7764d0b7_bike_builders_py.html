28#: Tags read to classify the infrastructure and the direction of travel (not kept as columns). 

29_CYCLE_TAGS = ("cycleway", "cycleway:left", "cycleway:right", "cycleway:both", "bicycle", "oneway", "oneway:bicycle") 

30#: Ways without motor traffic. 

31CAR_FREE = frozenset({"cycleway", "path", "footway", "pedestrian", "track", "bridleway"}) 

32#: Ways for pedestrians, used only with ``allow_footways`` or an explicit ``bicycle`` permission. 

33FOOTWAYS = frozenset({"footway", "pedestrian"}) 

34_QUIET = frozenset({"residential", "living_street", "unclassified", "service"}) 

35_TRACK = frozenset({"track", "opposite_track"}) 

36_LANE = frozenset({"lane", "opposite_lane", "share_busway", "opposite_share_busway"}) 

37_PERMITTED = frozenset({"yes", "designated", "permissive", "official"}) 

38_DEFAULT_SPEED_KMH = 40.0

41def _text(edges: pd.DataFrame, column: str) -> pd.Series: 

42 """A tag column as lower-case text, empty where the tag is absent.""" 

43 if column not in edges.columns: 

44 return pd.Series("", index=edges.index, dtype="string") 

45 return edges[column].astype("string").str.strip().str.lower().fillna("")

48def _highway_speed_kmh(highway: pd.Series) -> np.ndarray: 

49 """Default speed limit of each highway class, km/h (40 for classes outside ``HighwayType``).""" 

50 known = {item.value: item.max_speed * 0.06 for item in HighwayType} 

51 return highway.map(known).fillna(_DEFAULT_SPEED_KMH).to_numpy(dtype=float)

54def cycle_infrastructure(edges: pd.DataFrame) -> pd.Series: 

55 """Classify the cycling infrastructure of edges with OSM tags. 

56 

57 Returns a Series with ``"track"`` (a way of its own or without motor traffic: ``highway=cycleway``, 

58 paths and footways, or ``cycleway*=track``), ``"lane"`` (a marked lane on the carriageway: 

59 ``cycleway*=lane`` or a shared bus lane) or ``"mixed"`` (riding with the traffic). 

60 """ 

61 highway = _text(edges, "highway") 

62 tags = [_text(edges, column) for column in ("cycleway", "cycleway:left", "cycleway:right", "cycleway:both")] 

63 track = highway.isin(CAR_FREE) 

64 lane = pd.Series(False, index=edges.index) 

65 for tag in tags: 

66 track |= tag.isin(_TRACK) 

67 lane |= tag.isin(_LANE) 

68 return pd.Series(np.where(track, "track", np.where(lane, "lane", "mixed")), index=edges.index, name="cycle_infra")

71def level_of_traffic_stress(edges: pd.DataFrame) -> pd.Series: 

72 """Level of traffic stress (LTS) of edges with OSM tags, from 1 (anyone rides here) to 4 (only the fearless). 

73 

74 A simplified form of the criteria of Mekuria, Furth and Nixon (2012) for the tags OSM has: the kind of 

75 infrastructure (:func:`cycle_infrastructure`), the speed limit (``maxspeed``, or the default of the 

76 highway class) and the number of lanes (``lanes``, 2 when absent). 

77 

78 - track: 1; 

79 - lane: 1 up to 30 km/h and two lanes, 2 up to 50 km/h and two lanes, 3 up to 60 km/h or with more 

80 lanes, otherwise 4; 

81 - mixed traffic on residential, living, unclassified and service streets: 1 up to 30 km/h and two 

82 lanes (always on living streets), 2 up to 50 km/h and two lanes, 3 with more lanes, 4 above 50 km/h; 

83 - mixed traffic on other roads: 2 up to 30 km/h and two lanes, 3 up to 50 km/h and three lanes, 

84 otherwise 4. 

85 """ 

86 highway = _text(edges, "highway") 

87 infra = cycle_infrastructure(edges).to_numpy() 

88 limit = pd.to_numeric(edges["maxspeed"], errors="coerce") if "maxspeed" in edges.columns else pd.Series(np.nan) 

89 speed = np.where(np.isfinite(limit.to_numpy(dtype=float)), limit.to_numpy(dtype=float), _highway_speed_kmh(highway)) 

90 lanes = pd.to_numeric(edges["lanes"], errors="coerce") if "lanes" in edges.columns else pd.Series(np.nan) 

91 lanes = np.where(np.isfinite(lanes.to_numpy(dtype=float)), lanes.to_numpy(dtype=float), 2.0) 

92 quiet = highway.isin(_QUIET).to_numpy() 

93 living = (highway == "living_street").to_numpy() 

94 

95 lane_lts = np.select( 

96 [(speed <= 30) & (lanes <= 2), (speed <= 50) & (lanes <= 2), speed <= 60], 

97 [1, 2, 3], 

98 default=4, 

99 ) 

100 quiet_lts = np.select( 

101 [living | ((speed <= 30) & (lanes <= 2)), (speed <= 50) & (lanes <= 2), speed <= 50], 

102 [1, 2, 3], 

103 default=4, 

104 ) 

105 busy_lts = np.select([(speed <= 30) & (lanes <= 2), (speed <= 50) & (lanes <= 3)], [2, 3], default=4) 

106 lts = np.where(infra == "track", 1, np.where(infra == "lane", lane_lts, np.where(quiet, quiet_lts, busy_lts))) 

107 return pd.Series(lts.astype(np.int64), index=edges.index, name="lts")

116def get_bike_graph( # pylint: disable=too-many-locals 

117 *, 

118 osm_id: int | None = None, 

119 territory: Polygon | MultiPolygon | gpd.GeoDataFrame | None = None, 

120 simplify: bool = True, 

121 clip_by_territory: bool = False, 

122 keep_largest_subgraph: bool = True, 

123 bike_speed: float = 15 * 1000 / 60, # m/min 

124 max_lts: int | None = None, 

125 allow_footways: bool = False, 

126 elevation: str | os.PathLike | gpd.GeoDataFrame | Callable | None = None, 

127 network_type: Literal["bike", "custom"] = "bike", 

128 custom_filter: str | None = None, 

129 osm_edge_tags: list[str] | None = None, 

130) -> UrbanGraph: 

131 """ 

132 Build a cycling UrbanGraph from OpenStreetMap within a given territory. 

133 

134 The graph holds every way a bicycle may use: streets (except motorways and ways closed to bicycles), 

135 cycleways, paths and tracks, and footways and pedestrian streets where cycling is permitted 

136 (``bicycle=yes``, ``designated`` or ``permissive``). Every edge gets the kind of cycling infrastructure 

137 (``cycle_infra``: ``track``, ``lane`` or ``mixed``) and the level of traffic stress (``lts``, 1 to 4, see 

138 :func:`~iduedu.graph_builders.bike_builders.level_of_traffic_stress`). 

139 

140 The graph is directed. One-way streets are one-way for bicycles too, unless the way allows riding against 

141 the traffic (``oneway:bicycle=no`` or a ``cycleway*=opposite*`` tag). Travel time is the length over 

142 ``bike_speed``; with ``elevation`` it also depends on the grade. 

143 

144 Parameters: 

145 osm_id (int | None): OSM relation/area ID of the territory boundary. Provide this or ``territory``. 

146 territory (Polygon | MultiPolygon | gpd.GeoDataFrame | None): Boundary geometry in EPSG:4326 or GeoDataFrame. 

147 simplify (bool): If True, merges the segments of the ways into edges that run from junction to 

148 junction; segments are merged only when they carry the same tags. 

149 clip_by_territory (bool): If True, clips edges by the exact boundary geometry before graph construction. 

150 keep_largest_subgraph (bool): If True, returns only the largest strongly connected component. 

151 bike_speed (float): Cycling speed on flat ground in meters per minute. 

152 max_lts (int | None): Keep only edges with a level of traffic stress up to this one, for example 2 for 

153 the network that most adults are willing to ride. 

154 allow_footways (bool): If True, footways and pedestrian streets are used also without an explicit 

155 ``bicycle`` permission. 

156 elevation: Source of heights for :func:`~iduedu.add_elevation` (a raster path, a GeoDataFrame of points 

157 or contours, or a function). When given, edges get grades, and travel times are rescaled with the 

158 cycling speed model of :func:`~iduedu.apply_grade_speed`. 

159 network_type: ``"bike"``, or ``"custom"`` together with ``custom_filter``. 

160 custom_filter (str | None): Custom Overpass way filter used in custom mode. 

161 osm_edge_tags (list[str] | None): Which OSM tags to retain on edges. Overrides 

162 ``config.bike_useful_edges_attr`` (``highway``, ``name``, ``lanes``, ``maxspeed``, ``surface``). 

163 ``bridge`` and ``tunnel`` are always kept. The ``cycleway*``, ``bicycle``, ``oneway`` and 

164 ``oneway:bicycle`` tags are always read, and with ``simplify`` a change of any of them, or of a kept 

165 tag, starts a new edge. 

166 

167 Returns: 

168 UrbanGraph: Directed multigraph of the cycling network. Edges include geometry in the local CRS, 

169 ``length_meter``, ``time_min``, ``oneway``, ``cycle_infra``, ``lts``, the selected OSM tags and, with 

170 ``elevation``, ``grade``, ``grade_max``, ``ascent`` and ``descent``. 

171 

172 Raises: 

173 ValueError: If ``max_lts`` is not between 1 and 4, or the network type is unknown. 

174 """ 

175 if max_lts is not None and max_lts not in (1, 2, 3, 4): 

176 raise ValueError("max_lts must be 1, 2, 3, 4 or None") 

177 polygon4326 = get_4326_boundary(osm_id=osm_id, territory=territory) 

178 filters = {"bike": Network.BIKE.filter, "custom": custom_filter} 

179 try: 

180 road_filter = filters[network_type] 

181 except KeyError: 

182 raise ValueError(f"Unknown road_type: {network_type!r}") 

183 if network_type == "custom" and road_filter is None: 

184 raise ValueError("For road_type='custom' you must provide custom_filter") 

185 

186 needed_tags = set(config.bike_useful_edges_attr if osm_edge_tags is None else osm_edge_tags) 

187 tags_to_retrieve = needed_tags | set(_CYCLE_TAGS) | {"highway", "maxspeed", "lanes"} 

188 local_crs = estimate_crs_for_bounds(*polygon4326.bounds) 

189 

190 logger.info(f"Retrieving bike network via {osm_source_label()}...") 

191 edges = _build_edges_from_overpass( 

192 polygon4326, road_filter, needed_tags=tags_to_retrieve, local_crs=local_crs, simplify=simplify 

193 ) 

194 if len(edges) == 0: 

195 logger.warning("No edges found, returning empty graph") 

196 return _empty(local_crs, network_type) 

197 

198 highway, bicycle = _text(edges, "highway"), _text(edges, "bicycle") 

199 usable = bicycle != "no" 

200 if not allow_footways: 

201 usable &= ~highway.isin(FOOTWAYS) | bicycle.isin(_PERMITTED) 

202 edges = edges[usable.to_numpy()].reset_index(drop=True) 

203 if clip_by_territory and len(edges): 203 ↛ 204line 203 didn't jump to line 204 because the condition on line 203 was never true

204 clip_poly_gdf = gpd.GeoDataFrame(geometry=[polygon4326], crs=4326).to_crs(local_crs) 

205 edges = edges.clip(clip_poly_gdf, keep_geom_type=True).explode(ignore_index=True) 

206 edges["cycle_infra"] = cycle_infrastructure(edges) 

207 edges["lts"] = level_of_traffic_stress(edges) 

208 if max_lts is not None: 

209 edges = edges[edges["lts"] <= max_lts].reset_index(drop=True) 

210 if len(edges) == 0: 

211 logger.warning("No edges left for bicycles, returning empty graph") 

212 return _empty(local_crs, network_type) 

213 

214 # One-way for bicycles: the direction of the traffic, unless riding against it is allowed. 

215 oneway = _text(edges, "oneway") 

216 contraflow = (_text(edges, "oneway:bicycle") == "no").to_numpy() 

217 for column in ("cycleway", "cycleway:left", "cycleway:right", "cycleway:both"): 

218 contraflow |= _text(edges, column).str.startswith("opposite").to_numpy() 

219 reverse_mask = oneway.eq("-1").to_numpy() & ~contraflow 

220 if reverse_mask.any(): 

221 edges.loc[reverse_mask, "geometry"] = edges.loc[reverse_mask, "geometry"].apply( 

222 lambda geom: LineString(list(geom.coords)[::-1]) 

223 ) 

224 edges["oneway"] = oneway.isin({"yes", "true", "1", "-1"}).to_numpy() & ~contraflow 

225 

226 nodes, edges = _build_nodes_and_uv(edges, local_crs) 

227 edges["length_meter"] = edges.geometry.length.round(3) 

228 edges["time_min"] = (edges["length_meter"] / float(bike_speed)).round(3) 

229 edges["type"] = "bike" 

230 edges = _assign_edge_keys(edges) 

231 keep = needed_tags | set(STRUCTURE_TAGS) | {"u", "v", "k", "geometry", "length_meter", "time_min", "type"} 

232 keep |= {"oneway", "cycle_infra", "lts"} 

233 edges = edges[[column for column in edges.columns if column in keep]].copy() 

234 

235 graph = UrbanGraph( 

236 nodes_gdf=nodes, 

237 edges_gdf=edges, 

238 is_multigraph=True, 

239 is_directed=True, 

240 edge_direction_column="oneway", 

241 crs=local_crs, 

242 graph_type=network_type, 

243 ) 

244 if keep_largest_subgraph: 244 ↛ 245line 244 didn't jump to line 245 because the condition on line 244 was never true

245 graph = keep_largest_connected_component(graph) 

246 if elevation is not None: 

247 graph = apply_grade_speed(add_elevation(graph, elevation), "cycling") 

248 logger.debug("Bike graph built.") 

249 return graph
