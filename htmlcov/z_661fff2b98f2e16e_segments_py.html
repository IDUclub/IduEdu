11def segment_arrays(edges: gpd.GeoDataFrame, precision: float): 

12 """Unique straight segments of the edge geometries as arrays. 

13 

14 Returns ``(start, end, owner, counts)``: end point coordinates of each unique segment, the position in 

15 ``edges`` of the first edge that contains it, and how many edge pieces share it. Segments keep the edge 

16 order of the graph; zero-length segments are left out. 

17 """ 

18 coords, owner = shapely.get_coordinates(edges.geometry.values, return_index=True) 

19 same_edge = owner[1:] == owner[:-1] 

20 start, end, owner = coords[:-1][same_edge], coords[1:][same_edge], owner[:-1][same_edge] 

21 

22 # Grid integers make equal points compare equal; each segment is stored from its lower end point. 

23 a = np.round(start / precision).astype(np.int64) 

24 b = np.round(end / precision).astype(np.int64) 

25 swap = (b[:, 0] < a[:, 0]) | ((b[:, 0] == a[:, 0]) & (b[:, 1] < a[:, 1])) 

26 keys = np.where(swap[:, None], np.hstack([b, a]), np.hstack([a, b])) 

27 nonzero = np.any(keys[:, :2] != keys[:, 2:], axis=1) 

28 keys, start, end, owner = keys[nonzero], start[nonzero], end[nonzero], owner[nonzero] 

29 

30 _, first, counts = np.unique(keys, axis=0, return_index=True, return_counts=True) 

31 order = np.argsort(first) # keep the edge order of the graph 

32 first, counts = first[order], counts[order] 

33 return start[first], end[first], owner[first], counts

36def unique_segments(urban_graph: UrbanGraph, *, precision: float = 0.01) -> gpd.GeoDataFrame: 

37 """Return the straight segments the edge geometries are made of, each physical segment once. 

38 

39 Every pair of consecutive coordinates of an edge geometry is one segment. Segments are compared without 

40 direction and after rounding their end points to ``precision``, so the two directions of a two-way 

41 street, parallel edges drawn over the same points, and stretches that a there-and-back pair of edges 

42 shares only in part are all counted once. Separately mapped carriageways, which have their own points, 

43 stay separate. The sum of ``length`` is the length of the network on the ground. 

44 

45 Args: 

46 urban_graph: Graph with line geometries in a projected (metric) CRS. 

47 precision: Grid, in CRS units, that end points are rounded to before comparing segments. 

48 

49 Returns: 

50 GeoDataFrame in the graph CRS with one two-point LineString per unique segment and columns 

51 ``length``, ``edge`` (label of the first edge in ``edges_gdf`` that contains the segment) and 

52 ``edge_count`` (how many edge pieces share it). Zero-length segments are left out. 

53 

54 Raises: 

55 ValueError: If ``precision`` is not positive or the graph CRS is geographic. 

56 """ 

57 if not np.isfinite(precision) or precision <= 0: 

58 raise ValueError("precision must be a positive number") 

59 crs = projected_crs(urban_graph, "unique_segments") 

60 edges = urban_graph.edges_gdf 

61 columns = ["geometry", "length", "edge", "edge_count"] 

62 if edges.empty: 62 ↛ 63line 62 didn't jump to line 63 because the condition on line 62 was never true

63 return gpd.GeoDataFrame(columns=columns, geometry="geometry", crs=crs) 

64 

65 start, end, owner, counts = segment_arrays(edges, precision) 

66 geometry = shapely.linestrings(np.stack([start, end], axis=1)) 

67 return gpd.GeoDataFrame( 

68 { 

69 "length": shapely.length(geometry), 

70 "edge": edges.index.to_numpy()[owner], 

71 "edge_count": counts, 

72 }, 

73 geometry=geometry, 

74 crs=crs,
