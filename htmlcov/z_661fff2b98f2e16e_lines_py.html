1"""Cutting line geometries at distances along them, as ``shapely.ops.substring`` does, in linear time per line.""" 

2 

3import numpy as np 

4from shapely import LineString 

5 

6 

7def vertex_distances(coords: np.ndarray) -> np.ndarray: 

8 """Distance of every vertex from the start of the line, summed the way ``shapely.ops.substring`` sums it. 

9 

10 ``** 0.5`` per segment as in shapely, not ``np.sqrt``: the two can differ in the last bit, which decides 

11 whether a vertex lying exactly at a cut point is kept. 

12 """ 

13 xy = np.asarray(coords)[:, :2].tolist() 

14 lengths = [((x2 - x1) ** 2 + (y2 - y1) ** 2) ** 0.5 for (x1, y1), (x2, y2) in zip(xy, xy[1:])] 

15 return np.concatenate([[0.0], np.cumsum(lengths)])

18def cut_line(coords: np.ndarray, vertex_dist: np.ndarray, start: float, end: float, start_point, end_point): 

19 """``shapely.ops.substring(line, start, end)`` for ``0 <= start, end <= line.length``. 

20 

21 ``coords`` and ``vertex_dist`` (from :func:`vertex_distances`) describe the line once, so many cuts of one 

22 line cost a binary search each instead of a walk over the whole line. ``start_point`` and ``end_point`` are 

23 ``line.interpolate(start)`` and ``line.interpolate(end)``, the end points shapely uses. The result keeps the 

24 vertices strictly between the two distances, runs from ``start`` to ``end`` (backwards when 

25 ``start > end``) and is a two-point line when the distances are equal (where shapely returns a Point). 

26 """ 

27 first_xy = (start_point.x, start_point.y) 

28 if start == end: 

29 return LineString([first_xy, first_xy]) 

30 low, high = (start, end) if start < end else (end, start) 

31 first = np.searchsorted(vertex_dist, low, side="right") 

32 last = min(np.searchsorted(vertex_dist, high, side="left"), len(coords) - 1) 

33 inner = np.asarray(coords)[first:last, :2] 

34 if start > end: 

35 inner = inner[::-1] 

36 return LineString(np.vstack([[first_xy], inner, [(end_point.x, end_point.y)]]))
